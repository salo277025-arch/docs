 Load the UI Library Framework
local Rayfield = loadstring(game:HttpGet('https://githubusercontent.com'))()

-- 2. Create the main script hub window
local Window = Rayfield:CreateWindow({
   Name = "Blox Fruits AI Script Hub",
   LoadingTitle = "Initializing Delta Engine...",
   LoadingSubtitle = "AI Generated Hub",
   ConfigurationSaving = { Enabled = true, FolderName = "DeltaAI_Fruits" }
})

-- 3. Create Navigation Tabs
local FarmTab = Window:CreateTab("Auto Farm", 4483362458)
local TeleportTab = Window:CreateTab("Teleports", 4483362458)

-- 4. Feature 1: Auto Farm Toggle (AI Logic Integrated)
_G.AutoFarm = false
FarmTab:CreateToggle({
   Name = "Auto Farm Closest Enemies",
   CurrentValue = false,
   Callback = function(Value)
      _G.AutoFarm = Value
      task.spawn(function()
         while _G.AutoFarm do
            task.wait(0.1)
            pcall(function()
               local player = game.Players.LocalPlayer
               local character = player.Character
               if character and character:FindFirstChild("HumanoidRootPart") then
                  -- AI Logic: Find and gather enemies to the player
                  for _, enemy in pairs(workspace.Enemies:GetChildren()) do
                     if enemy:FindFirstChild("HumanoidRootPart") and enemy:FindFirstChild("Humanoid") and enemy.Humanoid.Health > 0 then
                        -- Bring enemy to player (Magnet Effect)
                        enemy.HumanoidRootPart.CFrame = character.HumanoidRootPart.CFrame + Vector3.new(0, 0, -4)
                        
                        -- Simulate basic attack click
                        local VirtualUser = game:GetService('VirtualUser')
                        VirtualUser:CaptureController()
                        VirtualUser:ClickButton1(Vector2.new(851, 529))
                     end
                  end
               end
            end)
         end
      end)
   end,
})

-- 5. Feature 2: Fruit Sniper/Teleport (AI Logic Integrated)
TeleportTab:CreateButton({
   Name = "Teleport to Spawned Fruit",
   Callback = function()
      local fruitFound = false
      for _, object in pairs(workspace:GetChildren()) do
         if string.find(object.Name, "Fruit") and object:FindFirstChild("Handle") then
            local playerObj = game.Players.LocalPlayer.Character
            if playerObj and playerObj:FindFirstChild("HumanoidRootPart") then
               playerObj.HumanoidRootPart.CFrame = object.Handle.CFrame
               fruitFound = true
               Rayfield:Notify({Title = "Fruit Found!", Content = "Teleported to " .. object.Name, Duration = 3})
               break
            end
         end
      end
      if not fruitFound then
         Rayfield:Notify({Title = "Status", Content = "No spawned fruits detected on the map.", Duration = 3})
      end
   end,
})
